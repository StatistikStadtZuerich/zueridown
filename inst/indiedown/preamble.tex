% create two minipages left and right aligned

% figures
\DeclareCaptionFormat{myformat_fig}{
\hspace{-3pt}\rule{\textwidth}{0.08em}
\begin{minipage}[t]{.70\linewidth}
\begin{flushleft}
#3 
\end{flushleft}
\end{minipage}
\hfill
\begin{minipage}[t]{.10\linewidth}
\begin{flushright}
#2#1
\end{flushright} 
\end{minipage}
\rule{\linewidth}{0.08em}
\vspace*{4mm}
}

% apply the new created format
\captionsetup[figure]{name=G -,format=myformat_fig, labelsep=space}
